1: #!/usr/bin/env node
2: import { readFileSync } from 'node:fs';
3: 
4: const die = message => { console.error(`stop-rule: ${message}`); process.exit(2); };
5: const object = value => value !== null && typeof value === 'object' && !Array.isArray(value);
6: const text = value => typeof value === 'string' && value.trim().length > 0;
7: let run;
8: try { run = JSON.parse(readFileSync(process.argv[2] || 0, 'utf8')); }
9: catch (error) { die(`Cannot read JSON: ${error.message}`); }
10: if (!object(run)) die('Input must be an object.');
11: if (!Array.isArray(run.in_scope) || !run.in_scope.length || run.in_scope.some(gate => !text(gate)) || new Set(run.in_scope).size !== run.in_scope.length) die('in_scope must contain unique non-empty gate names.');
12: const supportedGates = new Set(['G0', 'G1', 'G2', 'G3', 'G4', 'G5', 'G6', 'GP']);
13: if (run.in_scope.some(gate => !supportedGates.has(gate))) die('in_scope contains an unsupported gate.');
14: if (!Array.isArray(run.iterations) || !run.iterations.length) die('iterations must be non-empty.');
15: const scope = run.in_scope;
16: const gateList = (value, label) => {
17:   if (!Array.isArray(value) || value.some(gate => !scope.includes(gate)) || new Set(value).size !== value.length) die(`${label} must contain unique in-scope gates.`);
18: };
19: const reasons = (value, label) => {
20:   if (!object(value) || Object.entries(value).some(([gate, reason]) => !scope.includes(gate) || !text(reason))) die(`${label} must map in-scope gates to non-empty reasons.`);
21: };
22: if (run.not_applicable !== undefined) reasons(run.not_applicable, 'not_applicable');

25: let lastFix = 0;
26: const states = [];
27: for (const [index, iteration] of run.iterations.entries()) {
28:   if (!object(iteration) || !Number.isInteger(iteration.n) || iteration.n <= previousN) die(`Iteration ${index + 1} needs a strictly increasing positive integer n.`);
29:   previousN = iteration.n;
30:   if (!object(iteration.scores)) die(`Iteration ${iteration.n} needs a scores object.`);
31:   const blockedInput = iteration.infra_blocked === undefined ? [] : iteration.infra_blocked;
32:   const skippedInput = iteration.skipped === undefined ? {} : iteration.skipped;
33:   const naInput = iteration.not_applicable === undefined ? {} : iteration.not_applicable;

35:   gateList(blockedInput, `Iteration ${iteration.n} infra_blocked`);

37:   reasons(naInput, `Iteration ${iteration.n} not_applicable`);
38:   if (iteration.fix_applied !== undefined && typeof iteration.fix_applied !== 'boolean') die('fix_applied must be boolean.');
39:   if (iteration.g6_passed !== undefined && typeof iteration.g6_passed !== 'boolean') die('g6_passed must be boolean.');
40:   const na = { ...(run.not_applicable || {}), ...naInput };
41:   for (const [gate, reason] of Object.entries(na)) {
42:     if (/\b(unavailable|no tools?|missing tools?|not installed|infra(?:structure)? blocked|timeout)\b/i.test(reason)) die(`Non-applicability for ${gate} cannot be justified by tool absence.`);
43:     if (Object.hasOwn(skippedInput, gate) || blockedInput.includes(gate)) die(`${gate} cannot be both not applicable and skipped/blocked.`);
44:   }
45:   for (const [gate, score] of Object.entries(iteration.scores)) {
46:     if (!scope.includes(gate) || !Number.isInteger(score) || score < 0 || score > 5) die(`Iteration ${iteration.n}: scores must be integers 0..5 for in-scope gates.`);
47:     if (!iteration.measured.includes(gate) && measuredScores[gate] !== undefined && score !== measuredScores[gate]) die(`Iteration ${iteration.n}: score for ${gate} changed without measurement.`);
48:   }
49:   for (const gate of iteration.measured) {
50:     if (Object.hasOwn(skippedInput, gate) || blockedInput.includes(gate) || Object.hasOwn(na, gate)) die(`Iteration ${iteration.n}: ${gate} cannot be measured and omitted.`);
51:     if (gate === 'G6' && iteration.g6_passed !== undefined) measuredScores[gate] = iteration.g6_passed ? 5 : 0;
52:     else if (!Object.hasOwn(iteration.scores, gate)) die(`Iteration ${iteration.n}: measured ${gate} needs a score.`);
53:     else measuredScores[gate] = iteration.scores[gate];
54:     measuredAt[gate] = iteration.n;
55:   }
56:   // A fix lands after this iteration's observations, invalidating them all.
57:   if (iteration.fix_applied) lastFix = iteration.n;
58:   const required = scope.filter(gate => !Object.hasOwn(na, gate));
59:   const skipped = required.filter(gate => Object.hasOwn(skippedInput, gate));
60:   const blocked = required.filter(gate => blockedInput.includes(gate));
61:   const missing = required.filter(gate => measuredAt[gate] === undefined || measuredScores[gate] === undefined);
62:   const stale = required.filter(gate => measuredAt[gate] !== undefined && measuredAt[gate] <= lastFix);
63:   const failing = required.filter(gate => measuredScores[gate] !== undefined && measuredScores[gate] < 4);
64:   const covered = !skipped.length && !blocked.length && !missing.length && !stale.length;
65:   states.push({ iteration, required, skipped, blocked, missing, stale, failing, na, scores: { ...measuredScores }, covered });
66: }
67: const last = states.at(-1);
68: let stuck = 0, plateau = 0;
69: for (let index = 1; index < states.length; index++) {
70:   const now = states[index], before = states[index - 1];
71:   const shared = now.required.filter(gate => before.scores[gate] !== undefined && now.scores[gate] !== undefined);
72:   const improved = shared.some(gate => now.scores[gate] > before.scores[gate]) && !shared.some(gate => now.scores[gate] < before.scores[gate]);
73:   plateau = improved ? 0 : plateau + 1;
74:   const unchangedFailures = now.failing.length && now.failing.length === before.failing.length && now.failing.every(gate => before.failing.includes(gate) && now.scores[gate] <= before.scores[gate]);
75:   stuck = unchangedFailures ? stuck + 1 : 0;
76: }
77: const zeroed = last.required.filter(gate => last.scores[gate] === 0);
78: const coverage = last.covered ? 'COMPLETE' : last.required.length && last.blocked.length === last.required.length ? 'BLOCKED' : 'PARTIAL';
79: let decision = 'CONTINUE', verdict = last.failing.length ? 'DO-NOT-SHIP' : 'PENDING';
80: if (last.iteration.n > 6) { decision = 'STOP_CEILING'; verdict = zeroed.length || last.failing.length ? 'DO-NOT-SHIP' : 'BEST-EFFORT'; }
81: else if (last.covered && last.required.length && !last.failing.length && last.iteration.g6_passed !== false) { decision = 'SHIP'; verdict = 'SHIP'; }
82: else if (!last.required.length) { decision = 'NOT_APPLICABLE'; verdict = 'NOT-APPLICABLE'; }
83: else if (last.iteration.n >= 6) { decision = 'STOP_CEILING'; verdict = zeroed.length || last.failing.length ? 'DO-NOT-SHIP' : 'BEST-EFFORT'; }
84: else if (coverage === 'BLOCKED') decision = 'BLOCKED';
85: else if (plateau >= 3) { decision = 'STOP_PLATEAU'; verdict = last.failing.length ? 'DO-NOT-SHIP' : 'BEST-EFFORT'; }
86: else if (zeroed.length) { decision = 'FIX'; verdict = 'DO-NOT-SHIP'; }
87: else if (stuck >= 2) decision = 'CHANGE_APPROACH';
88: console.log(JSON.stringify({
89:   decision, verdict, coverage, iteration: last.iteration.n, ceiling: 6,
90:   surface_score: zeroed.length ? 0 : null,
91:   consecutive_fresh_passes: last.covered && !last.failing.length && last.required.length ? 1 : 0,
92:   passes_required: 1, lowest_failing_gate: [...last.failing].sort((left, right) => last.scores[left] - last.scores[right])[0] || null,
93:   failing_gates: last.failing, stale_gates: last.stale, missing_gates: last.missing,
94:   skipped: last.skipped, infra_blocked: last.blocked, not_applicable: last.na,
95:   stuck_count: stuck, no_improve_count: plateau,
96:   limit: 'Coverage describes observations. SHIP is a scoped review result, not deployment authorization. Non-applicability reasons require independent review.',
97: }, null, 2));
