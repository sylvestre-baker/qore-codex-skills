def launch(name,prompt,audit,timeout=900,label='turn1'):
 audit=Path(audit);(audit/(label+'-prompt.txt')).write_text(prompt+'\n')
 args=['docker','exec','-i','--workdir','/workspace/worker',name,'codex','--ask-for-approval','never','exec','--ignore-user-config','--ignore-rules','--ephemeral','--json','--skip-git-repo-check','--strict-config','-m',MODEL]
 for c in CONFIGS:args+=['-c',c]
 args+=['-']
 started=now();write(audit/(label+'-launch.json'),{'started_at':started,'argv':args,'prompt_sha256':sha(audit/(label+'-prompt.txt'))})
 with (audit/(label+'-stdout.jsonl')).open('w') as out,(audit/(label+'-stderr.txt')).open('w') as err:
  proc=subprocess.Popen(args,stdin=subprocess.PIPE,stdout=out,stderr=err,text=True)
  try:proc.communicate(prompt,timeout=timeout);exit_code=proc.returncode
  except subprocess.TimeoutExpired:
   proc.kill();proc.communicate();exit_code=124
 write(audit/(label+'-exit.json'),{'started_at':started,'ended_at':now(),'exit_code':exit_code})
 return exit_code

def collect(name,audit,label='final'):
 audit=Path(audit);archive=call(['docker','exec',name,'tar','-C','/workspace/output','-cf','-','.']).stdout
 (audit/(label+'-output.tar')).write_bytes(archive)
 dest=audit/(label+'-output');dest.mkdir(exist_ok=False)
 with tarfile.open(fileobj=io.BytesIO(archive)) as t:
  for m in t.getmembers():
   path=PurePosixPath(m.name.removeprefix('./'))
   if '..' in path.parts or path.is_absolute() or m.issym() or m.islnk():raise ValueError('Unsafe output member')
   if m.isfile():
    target=dest/path;target.parent.mkdir(parents=True,exist_ok=True);target.write_bytes(t.extractfile(m).read())
 return dest
